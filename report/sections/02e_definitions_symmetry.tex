\section{Definitions, Continued: Finite BC States and Covariant Generators}
\label{sec:definitions-symmetry}

\begin{definition}[Finite phase marginal and matrix extension]
\label{def:bc-finite-phase}
For an odd prime $\pp$, the phase algebra generated by $e(k/\pp)$ is
$C(\mathbb F_{\pp})$, represented diagonally on $H_{\pp}=\ltwo{\mathbb F_{\pp}}$.
Let $w_{\invtemp}$ be the restriction of the unique BC state for
$0<\invtemp\le1$, or of the Galois average of extremal BC states for
$\invtemp>1$. Its \emph{diagonal matrix extension} is
$\BCres{\pp,\invtemp}=\operatorname{diag}(w_{\invtemp}(x))$ in $B(H_{\pp})$.
A matrix extension is a density with these diagonal entries; its
off-diagonal entries are additional data. This does not specify a
representation of the full BC algebra on $H_{\pp}$.
\end{definition}

\begin{definition}[Local arithmetic covariance and parity]
\label{def:finite-arithmetic-covariance}
The centred Weyl operators $\Weyl{v}$, $v\in\mathbb F_{\pp}^2$, are those of
\cref{def:weyl-symplectic} with one register. Three requirements on a
superoperator are distinguished: commuting with $\Ad(U_a)$,
$U_a|x\rangle=|ax\rangle$, $a\in\mathbb F_{\pp}^{\times}$ (\emph{Galois torus});
commuting with $\Ad(\Weil(g))$, $g\in\Sp(2,\mathbb F_{\pp})$
(\emph{linear symplectic covariance}); and commuting with all
$\Ad(\Weyl{v})$ (\emph{Weyl-translation covariance}). The first action sends
$(x,y)$ to $(ax,a^{-1}y)$, and induces the inverse-labelled BC action on
the phase generators. Weyl-translation covariance is an extra condition,
not a consequence of carrying Weyl operators. Parity is
$\Par|x\rangle=|-x\rangle$; operator parity is $X\mapsto\Par X\Par$.
\end{definition}

\begin{definition}[Stationary covariant GKLS cone]
\label{def:stationary-covariant-cone}
For a density $\BCbond$ on $\mathbb C^d$ and a unitary group action $U$, let
$\Gcone_U(\BCbond)$ be the cone of Schr\"odinger generators of CPTP semigroups
with $\Lgen(\BCbond)=0$ and
$[\Lgen,\Ad(U_g)]=0$ for every $g$. Its dimension means the real
dimension of its linear span, including the overall time scale.
The identity is fixed by the dual Heisenberg semigroup; fixing $I$ also in
the Schr\"odinger picture is an additional condition, equivalent to
stationarity of the tracial density $I/d$.
\end{definition}

\begin{definition}[Conditional Choi matrix]
\label{def:generator-choi}
For a complex-linear map $L$ on $M_d$, set
\[
 \Choi_L=\sum_{i,j}|i\rangle\langle j|\otimes L(|i\rangle\langle j|),
 \qquad b=d^{-1/2}\sum_i|i,i\rangle,\qquad Q=I-|b\rangle\langle b|.
\]
The \emph{conditional Choi matrix} is $Q\Choi_LQ$.
The completely depolarizing channel is
$\Dep_d(X)=\Tr(X)I/d$.
\end{definition}
